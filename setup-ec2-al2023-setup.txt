# ffmpeg / ffprobe
# AL2023 の標準リポジトリには ffmpeg が無いため、静的ビルド (https://johnvansickle.com/ffmpeg/) を入れる
# x86_64 / aarch64 (Graviton) を自動判別し、md5 チェック後に /usr/local/bin へ配置する
echo 'ffmpeg / ffprobe をインストール...'
case "$(uname -m)" in
    x86_64)  FFMPEG_ARCH=amd64 ;;
    aarch64) FFMPEG_ARCH=arm64 ;;
    *)       FFMPEG_ARCH= ; echo "未対応のアーキテクチャ: $(uname -m)" ;;
esac
FFMPEG_TMP=$(mktemp -d)
cd $FFMPEG_TMP
if [ -n "$FFMPEG_ARCH" ] \
   && curl -fsSLO https://johnvansickle.com/ffmpeg/releases/ffmpeg-release-$FFMPEG_ARCH-static.tar.xz \
   && curl -fsSLO https://johnvansickle.com/ffmpeg/releases/ffmpeg-release-$FFMPEG_ARCH-static.tar.xz.md5 \
   && md5sum -c ffmpeg-release-$FFMPEG_ARCH-static.tar.xz.md5; then
    tar xf ffmpeg-release-$FFMPEG_ARCH-static.tar.xz
    sudo install -m 0755 ffmpeg-*-static/ffmpeg ffmpeg-*-static/ffprobe /usr/local/bin/
else
    # ダウンロードに失敗した場合は、このリポジトリに置いてある x86_64 版を使う
    echo '静的ビルドの取得に失敗したため、リポジトリ同梱版 (x86_64) をインストールします'
    curl -fsSL -o ffmpeg https://raw.githubusercontent.com/mir-aie/000S-amazon-linux2-setup/master/files/bin/al2023/ffmpeg/ffmpeg
    curl -fsSL -o ffprobe https://raw.githubusercontent.com/mir-aie/000S-amazon-linux2-setup/master/files/bin/al2023/ffmpeg/ffprobe
    sudo install -m 0755 ffmpeg ffprobe /usr/local/bin/
fi
cd ~
rm -rf $FFMPEG_TMP

============================================
結論：Webは今のまま apache ユーザーで動かし、artisan も apache ユーザーで実行する運用をおすすめします。 実行ユーザーを1つに揃えれば、パーミッションの衝突はなくなります。

対策1：実行ユーザーを apache に揃える（根本対策）

# 手動実行
sudo -u apache php artisan xxx

# ec2-user の ~/.bashrc にエイリアスを入れておくと楽です
alias art='sudo -u apache php /var/www/xxx/artisan'
次のものも apache で動かします。

スケジューラ：sudo crontab -u apache -e に * * * * * php /path/artisan schedule:run
キューワーカー：supervisor や systemd の設定で user=apache
デプロイ後の config:cache や view:cache なども sudo -u apache で実行する
対策2：うっかり ec2-user で実行しても壊れないようにする（保険）
グループ書き込みを許可しておくと、どちらのユーザーで作ったファイルも両方から書けるようになります。


sudo usermod -aG apache ec2-user
sudo chown -R apache:apache storage bootstrap/cache
sudo chmod -R ug+rwX storage bootstrap/cache
sudo find storage bootstrap/cache -type d -exec chmod g+s {} \;   # 新しいディレクトリにもグループ apache を引き継がせる
さらに、新しく作られるファイルがグループで書き込めるように umask を 002 にします。

php-fpm：sudo systemctl edit php-fpm で [Service] の下に UMask=0002
ec2-user：~/.bashrc に umask 002
このプロジェクトで気になった点
config/logging.php:145 で audit ログが 'permission' => 0644 になっています。このため、日付が変わって最初に書き込んだユーザーしかそのファイルに書けません。対策2を取る場合は 0664 に変更してください。他のチャンネルも含めて揃えておくと安心です。

まずは対策1（sudo -u apache に統一）だけでも、ご相談の問題はほぼ解消するはずです。

SMTPサーバと一括配信システムとの連携
※現状平塚市が prd-mms-01a/b-p82al2 に建てたSMTPでJ-ALERTと連携している。


0. Postfix のインストール（AL2023 には MTA が入っていない）

sudo dnf install -y postfix
sudo cp -p /etc/postfix/main.cf /etc/postfix/main.cf.bak.$(date +%Y%m%d)
# 外部 (J-ALERT) からの SMTP を受けるため、全インターフェースで待ち受け、受信ドメインを設定する
sudo postconf -e 'myhostname = smtp.a-alert.net'
sudo postconf -e 'mydomain = a-alert.net'
sudo postconf -e 'inet_interfaces = all'
sudo postconf -e 'mydestination = $myhostname, localhost.$mydomain, localhost, $mydomain'
sudo systemctl enable --now postfix
# セキュリティグループで送信元を J-ALERT 側に絞って 25/tcp を開けること

1. 受信ドメインの確認
transport_maps のキーは「ローカル部@ドメイン」の完全一致です。ドメイン名を正確に確認します。

postconf myhostname mydomain default_privs
smtp.a-alert.net が $myhostname か $mydomain のどちらかに一致していることを確認してください。

2. /etc/postfix/master.cf に追記

sudo cp -p /etc/postfix/master.cf /etc/postfix/master.cf.bak.$(date +%Y%m%d)
sudo vi /etc/postfix/master.cf
末尾に追記します。継続行は空白で始めてください。


jalert500001 unix  -       n       n       -       -       pipe
  flags=Rq user=ec2-user argv=/usr/bin/php /var/www/production/300L-demo01-500001-alert-v3/live/artisan jalert_smtp_receiver
jalert142034 unix  -       n       n       -       -       pipe
  flags=Rq user=ec2-user argv=/usr/bin/php /var/www/production/300L-hiratsuka-142034-alert-v3/live/artisan jalert_smtp_receiver


3. /etc/postfix/transport を作成

sudo vi /etc/postfix/transport

jalert-demo01-500001@smtp.a-alert.net      jalert500001:
jalert-hiratsuka-142034@smtp.a-alert.net   jalert142034:
t1@smtp.a-alert.net                        jalert500001:

sudo postmap /etc/postfix/transport
sudo postconf -e 'transport_maps = hash:/etc/postfix/transport'

4. /etc/aliases はそのまま残す

local_recipient_maps = proxy:unix:passwd.byname $alias_maps なので、alias の行を消すと宛先不明として受信拒否されます。transport_maps が優先されるため、alias の行が残っていても実行されなくなります。

行を残しつつ、万一の apache 実行を防ぎたい場合は、宛先を ec2-user に書き換えて sudo newaliases を実行します。


jalert-demo01-500001: ec2-user
jalert-hiratsuka-142034: ec2-user

5. 反映と確認

sudo postfix check
sudo systemctl reload postfix
postmap -q jalert-hiratsuka-142034@smtp.a-alert.net hash:/etc/postfix/transport   # → jalert142034:
postmap -q jalert-demo01-500001@smtp.a-alert.net hash:/etc/postfix/transport   # → jalert500001:

6. storage を ec2-user / apache の両方で書けるようにする

pipe は ec2-user で artisan を実行し、Web (php-fpm) は apache で同じ storage に書き込む。
所有者を ec2-user に chown すると Web 側が書けなくなるので、chown はせず laravel_shared_perms.sh で共有設定にする。
（master.cf の argv と同じディレクトリを指定すること）

for d in /var/www/production/300L-demo01-500001-alert-v3/live /var/www/production/300L-demo01-500001-alert-v3/test /var/www/production/300L-hiratsuka-142034-alert-v3/live /var/www/production/300L-hiratsuka-142034-alert-v3/test; do
  ~/bin/laravel_shared_perms.sh $d
done

7. テスト

echo -e "Subject: TEST\n\nTEST" | sendmail -f test@example.com jalert-demo01-500001@smtp.a-alert.net
sudo tail -f /var/log/maillog
maillog に relay=jalert500001, ... status=sent (delivered via jalert500001 service) が出れば成功です。
Laravel のログに [JALERT] run as uid=1000 user=ec2-user が出ることも確認してください。確認できたら、デバッグ用のログ行は削除してください。
