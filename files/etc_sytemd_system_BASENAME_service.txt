# /etc/systemd/system/mir-BASENAME.service

[Unit]
Description="mir-BASENAME"
# ネットワークが使える状態になってから実行する
After=network-online.target
Wants=network-online.target

[Service]
# 実行時のカレントディレクトリ
WorkingDirectory=/var/www/production/BASENAME/live/

# 実行するスクリプト
ExecStart=/usr/bin/php /var/www/production/BASENAME/live/artisan schedule:run

# apache 権限で実行する
User=apache
Group=apache
# 1回実行して終了する
Type=oneshot
