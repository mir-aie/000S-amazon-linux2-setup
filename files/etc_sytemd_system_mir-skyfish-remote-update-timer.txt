#
# Skyfish 本番プログラム更新スクリプト（mir-skyfish-remote-update.service）を毎分起動するタイマー
#
# https://documentation.suse.com/ja-jp/sle-micro/6.0/html/Micro-systemd-working-with-timers/index.html#systemd-timer-types-realtime
#
# /etc/systemd/system/mir-skyfish-remote-update.timer

[Unit]
Description="Run Skyfish host status every minutes"

[Timer]
# 毎日（土日を含む）毎分 01 秒に実行する
# 書式は 曜日 年-月-日 時:分:秒（曜日を省略すると毎日）
OnCalendar=*-*-* *:*:01

# 起動するサービス
Unit=mir-skyfish-remote-update.service

[Install]
# enable するとサーバー起動時にタイマーが自動で有効になる
WantedBy=timers.target

#  sudo vi mir-skyfish-remote-update.timer
#  sudo vi mir-skyfish-remote-update.service
#  sudo systemd-analyze verify /etc/systemd/system/mir-skyfish-remote-update.*
#  sudo systemctl daemon-reload
#  sudo systemctl enable mir-skyfish-remote-update.timer       # 初回のみ：起動時に自動で有効にする
#  sudo systemctl restart mir-skyfish-remote-update.timer
#  systemctl list-timers mir-skyfish-remote-update.timer       # 次回の実行予定を確認
