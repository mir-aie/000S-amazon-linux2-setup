# php を ec2-user で走らせたい場合

# 必要に応じて、crontabとsupervisordのapache を ec2-userにする

# apacheユーザーから ec2-user に切り替えたい場合
sudo cp /etc/php-fpm.d/www.conf /etc/php-fpm.d/www.conf.original
more /etc/php-fpm.d/www.conf.original
sudo sed -i "s/user = apache/user = ec2-user/" /etc/php-fpm.d/www.conf
sudo sed -i "s/group = apache/group = ec2-user/" /etc/php-fpm.d/www.conf

sudo systemctl reload php-fpm
sudo systemctl restart php-fpm
sudo systemctl enable php-fpm
sudo systemctl enabled php-fpm
sudo systemctl is-enabled php-fpm


# supervisord
sudo sed -i "s/user=apache/user=ec2-user/" /etc/supervisord/conf.d/*.conf
sudo /usr/local/bin/supervisorctl update

# crontab 
sudo -u apache crontab -l
ないようを
sudo -u ec2-user crontab -e
して、そちらにのコピー

sudo -u apache crontab -e
で中身を消す。
